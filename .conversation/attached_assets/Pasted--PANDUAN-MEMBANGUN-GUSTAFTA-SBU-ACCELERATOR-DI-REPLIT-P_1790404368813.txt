## 📋 FASE 1: PERSIAPAN & PEMILIHAN STACK

### Mengapa Stack Ini?
Untuk aplikasi yang membutuhkan **AI reasoning (Hermes)**, **PDF parsing**, **image metadata analysis**, dan **API integration**, stack terbaik adalah:

| Layer | Teknologi | Alasan |
|:---|:---|:---|
| **Backend + AI Engine** | **Python + FastAPI** | Ekosistem AI/LLM terkaya (LangChain, PyMuPDF, Pillow) |
| **Frontend** | **Next.js (React)** | Modern, cepat, SEO-friendly, mudah di-deploy |
| **Database** | **PostgreSQL (via Supabase)** | Gratis tier generous, ada auth built-in |
| **AI Orchestration** | **LangChain + OpenAI/Anthropic API** | Framework standar untuk Agentic AI |
| **File Storage** | **Supabase Storage** | Untuk upload dokumen SBU |
| **WhatsApp API** | **Fonnte / Wablas** (lokal) | Alternatif murah dari Twilio |

```python
"""
HERMES ORCHESTRATOR
Agen pusat yang mendelegasikan tugas ke agen spesialis
"""
from langchain_openai import ChatOpenAI
from langchain.agents import AgentExecutor, create_openai_tools_agent
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from app.agents.simpk_agent import simpk_tools
from app.agents.kbli_agent import kbli_tools
from app.agents.smap_agent import smap_tools
from app.agents.tkk_agent import tkk_tools

class HermesOrchestrator:
    def __init__(self):
        self.llm = ChatOpenAI(
            model="gpt-4o",  # atau "claude-3-5-sonnet" untuk Anthropic
            temperature=0.2,  # rendah agar akurat secara regulasi
        )
        self.agent = self._build_agent()
    
    def _build_agent(self):
        # Gabung semua tools dari agen spesialis
        all_tools = simpk_tools + kbli_tools + smap_tools + tkk_tools
        
        prompt = ChatPromptTemplate.from_messages([
            ("system", """
Kamu adalah HERMES, orchestrator AI untuk Gustafta SBU Accelerator.
Tugasmu: Mendiagnosa kebutuhan BUJK (Kontraktor/Konsultan) untuk 
permohonan baru atau perpanjangan SBU berdasarkan regulasi terbaru:
- PP 28/2025, Permen PU 6/2025, Kepdirjen 281/2026

ATURAN KERAS (WAJIB DIPATUHI):
1. SMAP untuk BUJK Kecil: HANYA terima Sertifikat Bimtek Anti-Korupsi
2. SMAP untuk BUJK Menengah/Besar: HANYA terima PANCEK KPK atau ISO 37001
3. Dokumen SMAP internal TIDAK PERNAH diterima → TOLAK tegas
4. Larangan rangkap usaha Konsultansi-Konstruksi adalah MUTLAK
5. Klarifikasi sewa alat di SIMPK: 5 hari kerja atau otomatis DITOLAK
6. Pengalaman 2 periode (6 tahun) wajib untuk perpanjangan

Selalu berikan output dalam format:
- Status: ✅ AMAN / ⚠️ PERHATIAN / ❌ BLOKIR
- Diagnosa: [penjelasan singkat]
- Rekomendasi: [langkah konkret]
- Referensi Pasal: [jika relevan]
            """),
            MessagesPlaceholder(variable_name="chat_history"),
            ("human", "{input}"),
            MessagesPlaceholder(variable_name="agent_scratchpad"),
        ])
        
        agent = create_openai_tools_agent(self.llm, all_tools, prompt)
        return AgentExecutor(agent=agent, tools=all_tools, verbose=True)
    
    async def run(self, query: str, chat_history: list = None):
        """Jalankan orchestrator dengan query user"""
        result = await self.agent.ainvoke({
            "input": query,
            "chat_history": chat_history or []
        })
        return result["output"]

# Singleton instance
hermes = HermesOrchestrator()
```

```python
"""
SMAP STRICT GATE AGENT
Validator ketat untuk dokumen anti-penyuapan
"""
from langchain_core.tools import tool
from typing import Optional
import json

# Knowledge base jalur SMAP
SMAP_RULES = {
    "Kecil": {
        "accepted": ["Sertifikat Bimtek", "Pelatihan Anti-Korupsi"],
        "rejected": ["ISO 37001", "PANCEK", "SOP Internal"],
        "komitmen_batas": "3 tahun"
    },
    "Menengah": {
        "accepted": ["PANCEK KPK", "ISO 37001:2016"],
        "rejected": ["SOP Internal", "Sertifikat Bimtek"],
        "komitmen_batas": "2 tahun"
    },
    "Besar": {
        "accepted": ["PANCEK KPK", "ISO 37001:2016"],
        "rejected": ["SOP Internal", "Sertifikat Bimtek"],
        "komitmen_batas": "1 tahun"
    }
}

@tool
def validate_smap_document(
    kualifikasi: str,
    document_type: str,
    document_name: str
) -> str:
    """
    Validasi apakah dokumen SMAP yang diupload sesuai regulasi.
    
    Args:
        kualifikasi: "Kecil", "Menengah", atau "Besar"
        document_type: Jenis dokumen yang diupload
        document_name: Nama file dokumen
    
    Returns:
        Status validasi dengan rekomendasi
    """
    rules = SMAP_RULES.get(kualifikasi)
    if not rules:
        return f"❌ Kualifikasi '{kualifikasi}' tidak valid"
    
    # Cek apakah dokumen diterima
    is_accepted = any(
        accepted.lower() in document_type.lower() 
        for accepted in rules["accepted"]
    )
    
    if is_accepted:
        return f"""
✅ DOKUMEN SMAP DITERIMA

Kualifikasi: {kualifikasi}
Jenis Dokumen: {document_type}
Status: VALID sesuai Kepdirjen 281/2026
Batas Komitmen: {rules['komitmen_batas']}

📋 Langkah selanjutnya:
1. Upload dokumen ke SIJK Terintegrasi
2. Catat tanggal upload untuk tracking komitmen
        """.strip()
    else:
        return f"""
❌ DOKUMEN SMAP DITOLAK

Kualifikasi: {kualifikasi}
Jenis Dokumen: {document_type}
Status: TIDAK VALID

🚫 ALASAN:
Berdasarkan Permen PU 6/2025 Lampiran 1 huruf A3 Poin L,
BUJK {kualifikasi} HANYA menerima:
{chr(10).join('• ' + x for x in rules['accepted'])}

Dokumen '{document_type}' TIDAK termasuk dalam daftar di atas.

💡 REKOMENDASI:
{f'Untuk {kualifikasi}, segera:' + chr(10) + 
 ('• Ikuti Bimtek Anti-Korupsi di lembaga terakreditasi' if kualifikasi == 'Kecil' 
  else '• Isi PANCEK di aplikasi KPK atau urus ISO 37001 dari lembaga terakreditasi KAN')}
        """.strip()

@tool
def check_smap_commitment_expiry(
    kualifikasi: str,
    tanggal_upload: str
) -> str:
    """
    Hitung masa berlaku komitmen SMAP berdasarkan tanggal upload.
    """
    from datetime import datetime, timedelta
    
    rules = SMAP_RULES.get(kualifikasi)
    if not rules:
        return "Kualifikasi tidak valid"
    
    # Parse tanggal
    try:
        upload_date = datetime.strptime(tanggal_upload, "%Y-%m-%d")
    except:
        return "Format tanggal harus YYYY-MM-DD"
    
    # Hitung expiry
    years = int(rules["komitmen_batas"].split()[0])
    expiry_date = upload_date + timedelta(days=years*365)
    
    # Hitung sisa hari
    today = datetime.now()
    remaining_days = (expiry_date - today).days
    
    if remaining_days <= 0:
        status = "❌ KADALUARSA - Segera perbarui!"
    elif remaining_days <= 90:
        status = f"⚠️ SISA {remaining_days} HARI - Siapkan perpanjangan"
    else:
        status = f"✅ AMAN - Masih {remaining_days} hari"
    
    return f"""
📅 TRACKING KOMITMEN SMAP

Kualifikasi: {kualifikasi}
Tanggal Upload: {tanggal_upload}
Batas Komitmen: {rules['komitmen_batas']}
Tanggal Kadaluarsa: {expiry_date.strftime('%Y-%m-%d')}
Status: {status}
    """.strip()

# Export tools untuk orchestrator
smap_tools = [validate_smap_document, check_smap_commitment_expiry]
```

```python
"""
SBU HEALTH SCORE ENDPOINT
Diagnosa 7 dimensi kesehatan SBU
"""
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional
from app.agents.orchestrator import hermes

router = APIRouter()

class DiagnosisRequest(BaseModel):
    nib: str
    kualifikasi: str  # Kecil/Menengah/Besar
    jenis_usaha: str  # Konstruksi/Konsultansi

class DiagnosisResponse(BaseModel):
    score: int
    status: str
    dimensions: dict
    recommendations: list
    ai_summary: str

@router.post("/health-score", response_model=DiagnosisResponse)
async def get_sbu_health_score(request: DiagnosisRequest):
    """
    Hitung skor kesehatan SBU (0-100) berdasarkan 7 dimensi
    """
    # Query ke Hermes untuk diagnosa komprehensif

    Lakukan diagnosa kesehatan SBU untuk:
    - NIB: {request.nib}
    - Kualifikasi: {request.kualifikasi}
    - Jenis Usaha: {request.jenis_usaha}
    
    Cek 7 dimensi:
    1. Masa berlaku SBU
    2. Status TKK (kadaluarsa/rangkap)
    3. Status alat di SIMPK
    4. Komitmen SMAP
    5. Pengalaman SIMPAN 2 periode
    6. LKUT & BO AHU
    7. Konversi KBLI 2020→2025
    
    Berikan skor 0-100 dan rekomendasi konkret.
    """

        ai_response = await hermes.run(query)
```

```tsx
                  <div className="font-medium">{value as string}</div>
                </div>
              ))}
            </div>

            <div className="bg-blue-50 p-4 rounded-lg">
              <h3 className="font-bold mb-2">💡 AI Recommendation:</h3>
              <p className="text-sm whitespace-pre-wrap">{result.ai_summary}</p>
            </div>
          </div>
        )}
      </div>
    </main>
  );
}
```

## 🗄️ FASE 6: DATABASE SCHEMA (Supabase)

Jalankan SQL ini di Supabase SQL Editor:

```sql
-- Tabel Users (BUJK)
CREATE TABLE bujk (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nib VARCHAR(13) UNIQUE NOT NULL,
  nama_perusahaan TEXT NOT NULL,
  kualifikasi VARCHAR(20) CHECK (kulifikasi IN ('Kecil', 'Menengah', 'Besar', 'Spesialis')),
  jenis_usaha VARCHAR(30) CHECK (jenis_usaha IN ('Konstruksi', 'Konsultansi')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabel Pengalaman (SIMPAN tracking)
CREATE TABLE pengalaman (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  bujk_id UUID REFERENCES bujk(id),
  nomor_registrasi VARCHAR(50),
  kbli_2020 VARCHAR(5),
  kbli_2025 VARCHAR(5),
  subklasifikasi VARCHAR(10),
  tanggal_selesai DATE,
  status_konversi VARCHAR(20)
);

-- Tabel Peralatan (SIMPK tracking)
CREATE TABLE peralatan (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  bujk_id UUID REFERENCES bujk(id),
  nomor_id_peralatan VARCHAR(50),
  jenis_alat TEXT,
  status_simPK VARCHAR(20),
  tanggal_klarifikasi_sewa TIMESTAMPTZ,
  is_exempt_k3 BOOLEAN DEFAULT FALSE
);

-- Tabel Dokumen SMAP
CREATE TABLE smap_documents (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  bujk_id UUID REFERENCES bujk(id),
  jenis_dokumen VARCHAR(50), -- PANCEK/ISO/Bimtek
  tanggal_upload DATE,
  tanggal_kadaluarsa DATE,
  is_valid BOOLEAN
);

-- Tabel Diagnosa (History)
CREATE TABLE diagnosis_history (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  bujk_id UUID REFERENCES bujk(id),
  score INTEGER,
  dimensions JSONB,
  recommendations JSONB,
  ai_summary TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Indexes untuk performa
CREATE INDEX idx_bujk_nib ON bujk(nib);
CREATE INDEX idx_pengalaman_bujk ON pengalaman(bujk_id);
CREATE INDEX idx_peralatan_bujk ON peralatan(bujk_id);
```
