<content>import { test, describe } from 'node:test'
import assert from 'node:assert/strict'
import { calculateAssessment, getTransitionInfo } from '../src/lib/sbu-rules.ts'

describe('SBU Rules - Transition Date Regression', () => {
  describe('When regulatory prerequisites are NOT met', () => {
    test('getTransitionInfo returns null when official letter is missing', () => {
      const info = getTransitionInfo({
        officialLetter: null,
        siJKStatus: 'unknown',
        verifiedExpertReview: null
      })

      assert.equal(info, null, 'Transition info must be null when prerequisites are not met')
    })

    test('getTransitionInfo returns null when SIJK status is unknown', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'unknown',
        verifiedExpertReview: null
      })

      assert.equal(info, null, 'Transition info must be null when SIJK status is unknown')
    })

    test('getTransitionInfo returns null when verified expert review is missing', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'ready',
        verifiedExpertReview: null
      })

      assert.equal(info, null, 'Transition info must be null when expert review is missing')
    })

    test('getTransitionInfo returns null when expert review is unverified', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'ready',
        verifiedExpertReview: {
          verified: false,
          expertId: 'expert-123',
          date: '2026-09-20'
        }
      })

      assert.equal(info, null, 'Transition info must be null when expert review is not verified')
    })

    test('getTransitionInfo NEVER uses Kepdirjen date as trigger', () => {
      // Even with a valid Kepdirjen date, without other prerequisites, no transition info
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-01-01', // Early date
          article20Paragraph2: true
        },
        siJKStatus: 'unknown',
        verifiedExpertReview: null
      })

      assert.equal(info, null, 'Kepdirjen date alone must NOT trigger transition')
      assert.equal(info?.transitionDate, undefined, 'No transition date should be set')
      assert.equal(info?.deadline, undefined, 'No deadline should be set')
    })

    test('calculateAssessment does not include transition deadline when prerequisites missing', () => {
      const assessment = calculateAssessment({
        companyName: 'PT Test',
        businessType: 'Konstruksi',
        annualRevenue: 5000000000,
        employeeCount: 50
      }, {
        officialLetter: null,
        siJKStatus: 'unknown',
        verifiedExpertReview: null
      })

      assert.ok(assessment.id, 'Assessment should be created')
      assert.equal(assessment.transitionDeadline, undefined, 'No transition deadline when prerequisites missing')
      assert.equal(assessment.transitionDate, undefined, 'No transition date when prerequisites missing')
      assert.equal(assessment.countdownDays, undefined, 'No countdown when prerequisites missing')
    })
  })

  describe('When ALL regulatory prerequisites ARE met', () => {
    test('getTransitionInfo returns valid info only when all conditions satisfied', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'ready',
        verifiedExpertReview: {
          verified: true,
          expertId: 'expert-123',
          qualification: 'Ahli Skema SBU',
          date: '2026-09-20',
          source: 'LSBU Resmi',
          approval: 'ST-123/LSBU/2026'
        }
      })

      assert.ok(info, 'Transition info should exist when all prerequisites met')
      assert.ok(info.transitionDate, 'Transition date should be set')
      assert.ok(info.deadline, 'Deadline should be set')
      assert.equal(info.officialLetterNumber, '281/KPTS/DK/2026')
      assert.equal(info.siJKStatus, 'ready')
      assert.equal(info.expertVerified, true)
    })

    test('calculateAssessment includes transition info when all prerequisites met', () => {
      const assessment = calculateAssessment({
        companyName: 'PT Test',
        businessType: 'Konstruksi',
        annualRevenue: 5000000000,
        employeeCount: 50
      }, {
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'ready',
        verifiedExpertReview: {
          verified: true,
          expertId: 'expert-123',
          qualification: 'Ahli Skema SBU',
          date: '2026-09-20',
          source: 'LSBU Resmi',
          approval: 'ST-123/LSBU/2026'
        }
      })

      assert.ok(assessment.id, 'Assessment should be created')
      assert.ok(assessment.transitionDeadline, 'Transition deadline should exist')
      assert.ok(assessment.transitionDate, 'Transition date should exist')
      assert.ok(typeof assessment.countdownDays === 'number', 'Countdown should be calculated')
    })
  })

  describe('Edge cases - Partial prerequisites', () => {
    test('Missing Article 20 Paragraph 2 reference blocks transition', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: false // Missing reference
        },
        siJKStatus: 'ready',
        verifiedExpertReview: {
          verified: true,
          expertId: 'expert-123',
          date: '2026-09-20'
        }
      })

      assert.equal(info, null, 'Transition blocked without Article 20(2) reference')
    })

    test('SIJK status "not_ready" blocks transition', () => {
      const info = getTransitionInfo({
        officialLetter: {
          number: '281/KPTS/DK/2026',
          date: '2026-09-15',
          article20Paragraph2: true
        },
        siJKStatus: 'not_ready',
        verifiedExpertReview: {
          verified: true,
          expertId: 'expert-123',
          date: '2026-09-20'
        }
      })

      assert.equal(info, null, 'Transition blocked when SIJK not ready')
    })
  })
})
